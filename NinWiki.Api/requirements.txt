# Dependencias Python do NinWiki.Api.
# Instale com: pip install -r requirements.txt (dentro do venv ativado)

fastapi==0.141.1             # framework web assincrono que expoe os endpoints da API
uvicorn[standard]==0.53.0    # servidor ASGI que roda o FastAPI (o [standard] inclui reload/websockets)
sqlalchemy[asyncio]==2.0.54  # ORM - unica forma de tocar no banco, nunca SQL puro
asyncpg==0.31.0              # driver assincrono de Postgres usado pelo SQLAlchemy
alembic==1.20.0              # gera e aplica migrations (versionamento do schema do banco)
pydantic==2.13.5             # validacao de dados (usado pelo FastAPI e pelo Settings)
pydantic-settings==2.15.0    # extensao do pydantic para carregar configuracoes do .env
python-dotenv==1.2.3         # le o arquivo .env (usado tambem pelo Alembic mais pra frente)
greenlet==3.5.6              # dependencia interna do SQLAlchemy para o modo assincrono
psycopg[binary]==3.2.13       # driver SINCRONO de Postgres, usado so' pelo Alembic (migrations rodam em modo sincrono; a API continua 100% assincrona via asyncpg). Usamos a versao 3 (nao psycopg2): o psycopg2 tem um bug conhecido no Windows que quebra ao decodificar mensagens de erro do Postgres localizadas em portugues. Pin em 3.2.13 (nao 3.2.4) porque a 3.2.4 nao tem build pra Python 3.14 - a 3.2.10 foi a primeira versao 3.2.x com build pra 3.14, ficamos na ultima da mesma linha 3.2 por seguranca (evita pular pra 3.3.x sem testar)
python-multipart==0.0.32     # exigido pelo FastAPI para receber upload de arquivo (UploadFile) - ver app/api/uploads.py
httpx==0.28.1                 # cliente HTTP assincrono - usado pra chamar a API da DeepL (ver app/core/translation.py)
Pillow==12.3.0                 # redimensiona imagem no upload (ver app/api/uploads.py) - evita imagem borrada/gigante quando a origem tem resolucao muito maior que o card onde ela aparece
PyJWT==2.15.0                 # assina/valida o JWT da sessao do painel (ver app/core/security.py)
